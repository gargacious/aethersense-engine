cmake_minimum_required(VERSION 3.20)
project(aethersense_engine VERSION 0.1.0 LANGUAGES CXX)
add_library(aethersense_engine src/engine.cpp)
target_compile_features(aethersense_engine PUBLIC cxx_std_20)
target_include_directories(aethersense_engine PUBLIC include third_party)
target_compile_options(aethersense_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BUILD_TESTING)
 add_executable(engine-tests tests/engine_tests.cpp)
 target_link_libraries(engine-tests PRIVATE aethersense_engine)
 add_test(NAME engine-contracts-dsp-tracking COMMAND engine-tests)
endif()
if(BUILD_TESTING)
 add_executable(engine-oracle-fixture tests/oracle_fixture.cpp)
 target_link_libraries(engine-oracle-fixture PRIVATE aethersense_engine)
endif()
target_sources(aethersense_engine PRIVATE src/fixtures.cpp)
if(BUILD_TESTING)
 add_executable(engine-fixture-tests tests/fixture_tests.cpp)
 target_link_libraries(engine-fixture-tests PRIVATE aethersense_engine)
 add_test(NAME engine-portable-golden-fixtures COMMAND engine-fixture-tests --check ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)
endif()

target_sources(aethersense_engine PRIVATE src/dsp_extensions.cpp)
if(BUILD_TESTING)

 target_link_libraries(dsp-extensions-tests PRIVATE aethersense_engine)

endif()
